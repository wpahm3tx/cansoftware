async function editViaServer(file, prompt) { const image = await shrinkImage(file); const response = await fetch("/api/edit-image", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ image, prompt }) }); const data = await response.json(); if (!response.ok) { throw new Error( [data.error, data.detail].filter(Boolean).join(" ") ); } if (!data.url) { throw new Error("Düzenlenmiş fotoğraf adresi bulunamadı."); } return data; }
